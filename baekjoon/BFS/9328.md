# [열쇠](https://www.acmicpc.net/problem/9328)

## 해결

외곽 위치인 경우이면서 벽이 아닌 위치를 모두 큐에 넣는다. 이때 문인지 아닌지의 여부는 중요하지 않다.  
BFS를 진행하면서 벽이 아닌 위치를 모두 큐에 넣는다. 이후 큐에서 좌표 정보를 꺼냈을 때, 해당 좌표 정보에 문이 있다면 다시 큐에 넣고 열쇠를 만나는 경우에는 연관된 모든 문을 지워버린다.

G1 난이도라고 하기에는 생각보다 쉽게 풀었던 문제. 관련한 문제가 있었던거같은데, 기억이 안남.

다만 처음 문제를 풀 때, 외곽에 키가 있을 때 처리하지 않고, 그냥 입력시 키를 만나면 처리하는 로직으로 구현을 해버려서 fail이 나긴 했다. 이를 해결하는데 도움을 준 [TC](https://www.acmicpc.net/board/view/117173). TC 결과를 출력해보니 차단 되어있어야 할 문이 열려있는 것을 보고 문제를 파악, 해결했다.

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <unordered_map>

#define endl "\n"

using namespace std;

using item_t = char;
using line_t = vector<item_t>;
using board_t = vector<line_t>;

using Point2D = pair<int, int>;

constexpr item_t WALL = '*';
constexpr item_t EMPTY = '.';
constexpr item_t PAPER = '$';

constexpr char ZERO_KEY = '0';

constexpr item_t VISITED = 1;

static int offsets[][2] = {
	{0,1},
	{1,0},
	{-1,0},
	{0,-1}
};

bool range_validation(int x, int y, int w, int h) {
	if (x < 0 || x >= w) return false;
	if (y < 0 || y >= h) return false;

	return true;
}

void solution() {
	int h, w; cin >> h >> w;

	board_t board(h, line_t(w));
	board_t visited(h, line_t(w));

	unordered_map<item_t, vector<Point2D>> door_location;
	queue<Point2D> q;
	string key_list;

	for (int y = 0; y < h; ++y) {
		for (int x = 0; x < w; ++x) {
			cin >> board[y][x];
			if (board[y][x] == WALL) continue;

			if (isalpha(board[y][x]) && board[y][x] < 'a') { // door
				door_location[board[y][x]].emplace_back(x, y);
			}

			if (y == 0 || x == 0 || y == h - 1 || x == w - 1) { // aside
				if (isalpha(board[y][x]) && board[y][x] >= 'a') { // key
					key_list += board[y][x];
				}
				q.emplace(x, y);
			}
		}
	}

	// key initialize
	{
		string keyinput;
		cin >> keyinput;
		if (keyinput[0] == ZERO_KEY) {
			key_list = "";
		}
		else {
			key_list += keyinput;
		}

		for (auto key : key_list) {
			key ^= 0x20; // find connected door
			if (!door_location.count(key)) continue;

			for (auto [dx, dy] : door_location[key]) {
				board[dy][dx] = EMPTY;
			}
		}
	}

	int get_paper_count = 0;
	int loop = 0;
	while (q.size() && loop <= q.size()) {
		auto [x, y] = q.front(); q.pop();

		if (isalpha(board[y][x])) { // door || key
			if (board[y][x] < 'a') {
				q.emplace(x, y);
				loop++;
				continue;
			}
			else
			{
				for (auto [dx, dy] : door_location[board[y][x] ^ 0x20]) {
					board[dy][dx] = EMPTY;
				}
				loop = 0;
			}
		}

		if (board[y][x] == PAPER) {
			get_paper_count++;
			board[y][x] = EMPTY;
		}

		for (int i = 0; i < 4; ++i) {
			auto [cx, cy] = offsets[i];
			cx += x; cy += y;
			if (!range_validation(cx, cy, w, h)) continue;
			if (board[cy][cx] == WALL) continue;
			if (visited[cy][cx]) continue;
			visited[cy][cx] = VISITED;

			q.emplace(cx, cy);
			loop = 0;
		}

	}

	cout << get_paper_count << endl;

	//for (const auto& cntr : board) {
	//	for (const auto item : cntr)
	//		cout << item << " ";
	//	cout << endl;
	//}


}

int main() {
	cin.tie(0)->sync_with_stdio(0);
	cout.tie(0);

	int T;
	cin >> T;

	while (T--) solution();
}
```
