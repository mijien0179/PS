# [소문난 칠공주](https://www.acmicpc.net/problem/1941)

`백트래킹` 문제로 들어갔으나, 코드 흐름을 봤을 때 백트래킹보다는 브루트포스에 가까운 풀이가 필요했던 문제라 생각한다.

## 해결
- 순열 만들듯이 7개를 선택하고
- 선택된 7개가 연결되어있는지 보고
- 연결되어 있다면 그중 4개가 S인지

이렇게 코드를 구성했다.

처음 코드를 구성했을 때는 단일 차원 `vector`를 이용하려 했으나, 뭐가 이상한지 제대로 처리가 이뤄지지 않았다. 그래서 아래 코드처럼 구현했는데, 시간 나면 기존 고려했던 방법대로 코드를 다시 작성해봐야겠다.  
[0ms 대인 코드](https://www.acmicpc.net/source/88247817)도 있던데 그건 좀 분석좀 해보고.


```cpp
#include <iostream>
#include <vector>
#include <queue>

#define endl "\n"

using namespace std;

using item_t = char;
using line_t = vector<item_t>;
using board_t = vector<line_t>;

using Point2D = pair<int, int>;

constexpr int WIDTH = 5;
constexpr int HEIGHT = 5;

constexpr item_t SOM = 'S';
constexpr item_t YEON = 'Y';

bool range_validation(int x, int y) {
	if (x < 0 || x >= WIDTH) return false;
	if (y < 0 || y >= HEIGHT) return false;

	return true;
}

static vector<Point2D> offsets = {
	{0,-1},
	{-1,0},
	{0,1},
	{1,0}
};

int bt(board_t& d, board_t& visited, const int _x, const int _y, int depth) {
	int answer = 0;

	if (depth == 7) {
		queue<Point2D> q;
		vector<Point2D> rollback;
		q.emplace(_x, _y);
		visited[_y][_x] = 2;

		int total_count = 1;
		int count_of_som = (d[_y][_x] == SOM);
		while (q.size()) {
			auto [x, y] = q.front(); q.pop();

			for (auto [cx, cy] : offsets) {
				cx += x; cy += y;
				if (!range_validation(cx, cy)) continue;
				if (visited[cy][cx] != 1) continue;
				visited[cy][cx] = 2;
				total_count++;
				count_of_som += (d[cy][cx] == SOM);
				q.emplace(cx, cy);
				rollback.emplace_back(cx, cy);
			}
		}

		for (const auto& [x, y] : rollback) {
			visited[y][x] = 1;
		}

		if (total_count != 7) return 0;
		if (count_of_som < 4) return 0;

		return 1;
	}
	int x = _x+1;
	for (int y = _y; y < HEIGHT; ++y) {
		for (; x < WIDTH; ++x) {
			if (visited[y][x]) continue;
			visited[y][x] = 1;
			answer += bt(d, visited, x, y, depth + 1);
			visited[y][x] = 0;
		}
		x = 0;
	}

	return answer;
}

int main() {
	cin.tie(0)->sync_with_stdio(0);
	cout.tie(0);

	board_t board(HEIGHT, line_t(WIDTH)), visited(HEIGHT, line_t(WIDTH));
	for (auto& cntr : board) {
		for (auto& item : cntr)
			cin >> item;
	}

	int answer = bt(board, visited, -1, 0, 0);

	cout << answer << endl;
}
```
