# [ABCDE](https://www.acmicpc.net/problem/13023)

depth가 4까지 내려갈 수 있는 탐색이 있는가를 분별하는 문제. 처음 문제를 보고 이게 무슨 설명인가 했다.

## 해결

그래프 탐색을 좀 해봤다면 어렵지 않게 풀 수 있는 문제라고 생각한다. 나는 첫 진입점에 visited 값을 변경해주지 않아 17%에서 Fail을 받았다. dfs 진입 전에 꼭 visited 값 변경 하자.

```cpp
#include <iostream>
#include <vector>

using namespace std;

bool dfs(vector<vector<int>>& graph, vector<int>& visited, int start, int before, int depth) {
	if (depth == 0) return true;

	for (const auto& e : graph[start]) {
		if (e == before) continue;
		if (visited[e]) continue;

		visited[e] = 1;
		auto res = dfs(graph, visited, e, start, depth - 1);
		visited[e] = 0;
		if (res) return true;
	}

	return false;
}

int main() {
	int N, M;
	cin >> N >> M;

	vector<vector<int>> graph(N);

	for (int i = 0; i < M; ++i) {
		int s, e;
		cin >> s >> e;

		graph[s].push_back(e);
		graph[e].push_back(s);
	}

	vector<int> visited(N);
	bool res = 0;

	for (int i = 0; i < N; ++i) {
		fill(visited.begin(), visited.end(), 0);

		visited[i] = 1;
		res = dfs(graph, visited, i, -1, 4);
		if (res) break;
	}

	cout << res << endl;
}
```
